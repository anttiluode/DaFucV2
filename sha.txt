import torch
import torch.nn as nn
import torch.nn.functional as F
import math
import tkinter as tk

from typing import List, Dict, Optional, Tuple
import hashlib  # For computing SHA-256

# -------------------------------
# FieldPosition: Describes a unit's position and environmental info.
# -------------------------------
class FieldPosition:
    def __init__(self, physical_coords: np.ndarray, field_type: str, field_strength: float, gradients: Dict[str, float]):
        self.physical_coords = physical_coords  # e.g., a 2D coordinate (for visualization, also used as location in the field)
        self.field_type = field_type            # e.g., "sensory", "memory", etc.
        self.field_strength = field_strength    # local field intensity
        self.gradients = gradients              # gradients that bias specialization

# -------------------------------
# SmoothField: Maintains a smoothly updating 2D field.
# -------------------------------
class SmoothField:
    def __init__(self, field_size: int, update_rate: float = 0.1):
        self.field_size = field_size
        self.update_rate = update_rate
        self.field = np.zeros((field_size, field_size))
    
    def update(self, new_field: np.ndarray):
        # Exponential smoothing update
        self.field = (1 - self.update_rate) * self.field + self.update_rate * new_field
    
    def get_field(self) -> np.ndarray:
        return self.field

# -------------------------------
# SHA256Task: Generates a random binary problem and its SHA-256 hash.
#
# For our purposes, the “problem” is a random binary vector (of dimension input_dim)
# that is converted to a string (e.g. "101001..."). Its SHA-256 hash is computed,
# then converted into a binary string of length 256 which is reshaped into a 16×16 field.
# -------------------------------
class SHA256Task:
    def __init__(self, field_size: int = 16, input_dim: int = 10):
        self.field_size = field_size  # Should be chosen so that field_size*field_size == 256 for SHA-256.
        self.input_dim = input_dim

    def generate_target(self) -> Tuple[torch.FloatTensor, torch.FloatTensor]:
        # Generate a random binary vector of length input_dim (features)
        features = torch.randint(0, 2, (self.input_dim,), dtype=torch.float)
        # Convert features to a binary string (e.g. "1010010101")
        feature_str = ''.join(str(int(b.item())) for b in features)
        # Compute SHA-256 hash of the string (as bytes)
        hash_obj = hashlib.sha256(feature_str.encode('utf-8'))
        hash_hex = hash_obj.hexdigest()  # 64 hex characters (256 bits)
        # Convert hex to binary string (each hex digit = 4 bits)
        hash_bin = bin(int(hash_hex, 16))[2:].zfill(256)
        # Convert binary string into a list of floats (0.0 or 1.0)
        hash_bits = [float(bit) for bit in hash_bin]
        # Convert to a tensor and reshape into (field_size, field_size)
        target_field = torch.tensor(hash_bits, dtype=torch.float).view(self.field_size, self.field_size)
        return features, target_field

# -------------------------------
# AdaptiveUnit: Modified to accept input and output dimensions.
# -------------------------------
class AdaptiveUnit:
    def __init__(self, initial_position: FieldPosition, input_dim: int = 10, output_dim: int = 256):
        self.position = initial_position
        self.role = "undifferentiated"
        self.neighbors: List['AdaptiveUnit'] = []
        self.plasticity = 1.0
        self.max_layers = 10
        # For SHA field control, output_dim = 256 (16x16)
        self.network = nn.Sequential(
            nn.Linear(input_dim, 128),
            nn.ReLU(),
            nn.Linear(128, output_dim)
        )
        self.recent_losses: List[float] = []
        self.best_score = 0.0
        self.specialization_score: Dict[str, float] = {}
        self.role_functions = {
            "sensory": self._process_sensory,
            "memory": self._process_memory,
            "integration": self._process_integration,
            "output": self._process_output
        }
        self._memory_buffer: List[torch.Tensor] = []
        self._last_layer_addition = -10

    def compute_temporal_correlation(self) -> float:
        if len(self.recent_losses) < 2:
            return 0.0
        x = np.array(self.recent_losses[-2:])
        if np.all(x == x[0]):
            return 0.0
        corr = np.corrcoef([0, 1], x)[0, 1]
        return 0.0 if np.isnan(corr) else corr

    def sense_environment(self) -> Dict:
        local_gradients = self.position.gradients.copy()
        neighbor_roles = [n.role for n in self.neighbors]
        role_counts = {role: neighbor_roles.count(role) for role in set(neighbor_roles)}
        for role, count in role_counts.items():
            local_gradients[role] = local_gradients.get(role, 0) - (count * 0.1)
        return {
            "field_type": self.position.field_type,
            "field_strength": self.position.field_strength,
            "neighbor_states": neighbor_roles,
            "gradients": local_gradients
        }

    def adapt_to_field(self, field_data: Dict):
        if self.role == "undifferentiated":
            epoch_penalty = len(self.recent_losses) * 0.01
            for k in field_data["gradients"]:
                field_data["gradients"][k] += epoch_penalty
        temporal_correlation = self.compute_temporal_correlation()
        memory_boost = 0.2 * abs(temporal_correlation)
        affinities = {
            "sensory": field_data["gradients"].get("sensory", 0),
            "memory": field_data["gradients"].get("memory", 0) + memory_boost,
            "integration": field_data["field_strength"],
            "output": field_data["gradients"].get("output", 0)
        }
        best_role, best_affinity = max(affinities.items(), key=lambda x: x[1])
        if best_affinity > 0.8 and self.plasticity > 0.5:
            self.role = best_role
            self.plasticity *= 0.9
            if self.role == "memory":
                self._expand_memory_capacity()
            elif self.role == "integration":
                self._add_integration_layers()

    def process_input(self, input_data: torch.Tensor) -> torch.Tensor:
        if self.role in self.role_functions:
            return self.role_functions[self.role](input_data)
        return self.network(input_data)

    def _process_sensory(self, input_data: torch.Tensor) -> torch.Tensor:
        return torch.relu(self.network(input_data))

    def _process_memory(self, input_data: torch.Tensor) -> torch.Tensor:
        self.memory_buffer_append(input_data)
        if self._memory_buffer:
            mem_tensor = torch.stack(self._memory_buffer).mean(dim=0)
            return self.network(input_data + mem_tensor)
        return self.network(input_data)

    def _process_integration(self, input_data: torch.Tensor) -> torch.Tensor:
        own_output = self.network(input_data)
        neighbor_outputs = []
        for n in self.neighbors:
            if n.role == "integration":
                neighbor_outputs.append(n.network(input_data))
            else:
                neighbor_outputs.append(n.process_input(input_data))
        if neighbor_outputs:
            combined = torch.stack(neighbor_outputs).mean(dim=0)
            return own_output + combined
        return own_output

    def _process_output(self, input_data: torch.Tensor) -> torch.Tensor:
        raw_output = self.network(input_data)
        return F.softmax(raw_output, dim=0)

    def _expand_memory_capacity(self):
        print("Expanding memory capacity for unit at", self.position.physical_coords)

    def _add_integration_layers(self):
        current_step = len(self.recent_losses)
        if current_step - self._last_layer_addition < 2:
            return
        print("Adding integration layers for unit at", self.position.physical_coords)
        self._last_layer_addition = current_step

    def _enhance_role_specific_circuits(self):
        current_layers = list(self.network.children())
        if len(current_layers) >= self.max_layers:
            print(f"Complexity limit reached for unit at {self.position.physical_coords}. No further enhancement.")
            return
        if self.specialization_score.get(self.role, 0) < self.best_score * 1.05:
            return
        print(f"Enhancing {self.role} circuits for unit at {self.position.physical_coords}")
        new_layer = nn.Linear(1, 1)  # Output is 1D
        self.network = nn.Sequential(*current_layers, new_layer, nn.ReLU())

    def learn_from_experience(self, input_data: torch.Tensor, target: Optional[torch.Tensor] = None) -> Optional[float]:
        if self.plasticity > 0.1 and target is not None:
            role_factors = {
                "undifferentiated": 1.0,
                "memory": 0.8,
                "sensory": 1.2,
                "integration": 1.0,
                "output": 1.5
            }
            role_factor = role_factors.get(self.role, 1.0)
            current_complexity = len(list(self.network.children()))
            adaptive_lr = (0.01 * role_factor) / math.sqrt(current_complexity)
            optimizer = torch.optim.Adam(self.network.parameters(), lr=adaptive_lr)
            output = self.process_input(input_data)
            loss = nn.MSELoss()(output, target)
            optimizer.zero_grad()
            loss.backward()
            optimizer.step()
            loss_value = loss.item()
            self.recent_losses.append(loss_value)
            if len(self.recent_losses) > 10:
                self.recent_losses.pop(0)
            self.specialization_score[self.role] = self.specialization_score.get(self.role, 0) + (1 - loss_value)
            self.influence_neighbors()
            self.reinforce_specialization()
            return loss_value
        return None

    def influence_neighbors(self):
        if not self.neighbors or not self.recent_losses:
            return
        avg_loss = sum(self.recent_losses) / len(self.recent_losses)
        success_rate = math.exp(-avg_loss)
        influence_factor = 0.1 * (1 - len(self.neighbors) * 0.1)
        influence_factor = max(0, influence_factor)
        if success_rate > 0.8:
            for neighbor in self.neighbors:
                if neighbor.plasticity > 0.2:
                    for param, neighbor_param in zip(self.network.parameters(), neighbor.network.parameters()):
                        neighbor_param.data += influence_factor * (param.data - neighbor_param.data)

    def reinforce_specialization(self):
        if self.role != "undifferentiated":
            recent_success = self.specialization_score.get(self.role, 0)
            if recent_success > self.best_score:
                self.best_score = recent_success
                epoch_factor = len(self.recent_losses) / 20.0
                self.plasticity *= (0.97 + 0.03 * epoch_factor)
                self._enhance_role_specific_circuits()

    def freeze_state(self) -> Dict:
        self.plasticity = 0
        return {
            'role': self.role,
            'network_state': self.network.state_dict(),
            'specialization': self.specialization_score
        }

    def load_state(self, state: Dict):
        self.role = state['role']
        self.network.load_state_dict(state['network_state'])
        self.specialization_score = state['specialization']

    def memory_buffer_append(self, tensor: torch.Tensor):
        self._memory_buffer.append(tensor)
        if len(self._memory_buffer) > 100:
            self._memory_buffer.pop(0)

# -------------------------------
# UnitSystem: Manages multiple AdaptiveUnits.
# -------------------------------
class UnitSystem:
    def __init__(self):
        self.units: List[AdaptiveUnit] = []

    def add_unit(self, unit: AdaptiveUnit):
        for other in self.units:
            distance = np.linalg.norm(unit.position.physical_coords - other.position.physical_coords)
            if distance < 5.0:
                unit.neighbors.append(other)
                other.neighbors.append(unit)
        self.units.append(unit)

    def run_epoch(self, input_data: torch.Tensor, target: torch.Tensor) -> List[float]:
        losses = []
        for unit in self.units:
            field_data = unit.sense_environment()
            unit.adapt_to_field(field_data)
            loss = unit.learn_from_experience(input_data, target)
            if loss is not None:
                losses.append(loss)
        return losses

    def predict(self, input_data: torch.Tensor) -> torch.Tensor:
        outputs = []
        for unit in self.units:
            outputs.append(unit.process_input(input_data))
        return torch.mean(torch.stack(outputs), dim=0)

    def freeze_all(self) -> List[Dict]:
        return [unit.freeze_state() for unit in self.units]

    def load_all(self, state_list: List[Dict]):
        for unit, state in zip(self.units, state_list):
            unit.load_state(state)

# -------------------------------
# AdaptiveFieldController: Uses the UnitSystem to control a 2D field.
#
# (Here, the “field” represents a 16×16 grid corresponding to a SHA-256 hash.)
# -------------------------------
class AdaptiveFieldController:
    def __init__(self, field_size: int = 16, n_units: int = 10, input_dim: int = 10):
        self.field_size = field_size
        self.input_dim = input_dim
        self.output_dim = field_size * field_size  # 16*16 = 256
        self.smooth_field = SmoothField(field_size=field_size, update_rate=0.1)
        self.system = UnitSystem()
        for i in range(n_units):
            pos = FieldPosition(
                physical_coords=np.array([np.random.uniform(0, field_size), np.random.uniform(0, field_size)]),
                field_type=np.random.choice(['sensory', 'memory', 'integration', 'output']),
                field_strength=np.random.uniform(0, 1),
                gradients={
                    'sensory': np.random.uniform(0, 1),
                    'memory': np.random.uniform(0, 1),
                    'output': np.random.uniform(0, 1)
                }
            )
            unit = AdaptiveUnit(pos, input_dim=input_dim, output_dim=self.output_dim)
            self.system.add_unit(unit)

    def update(self, input_features: torch.Tensor) -> np.ndarray:
        output = self.system.predict(input_features)
        field_update = output.view(self.field_size, self.field_size).detach().numpy()
        self.smooth_field.update(field_update)
        return self.smooth_field.get_field()

    def train_step(self, input_features: torch.Tensor, target_field: torch.Tensor) -> float:
        target_flat = target_field.view(-1)
        losses = self.system.run_epoch(input_features, target_flat)
        return np.mean(losses) if losses else 0.0

    def test_step(self, input_features: torch.Tensor, target_field: torch.Tensor) -> float:
        target_flat = target_field.view(-1)
        output = self.system.predict(input_features)
        loss = nn.MSELoss()(output, target_flat)
        return loss.item()

# -------------------------------
# AdaptiveFieldVisualizerApp: Tkinter GUI to visualize field control and network dynamics.
#
# In this version we add a toggle button to switch between Training mode (where the SHA problem
# and its known solution are used to train the network) and Testing mode (where the network
# is not updated and its prediction for a new SHA problem is evaluated against the true hash).
# -------------------------------
class AdaptiveFieldVisualizerApp:
    def __init__(self, root, controller: AdaptiveFieldController, task: SHA256Task):
        self.root = root
        self.controller = controller
        self.task = task
        self.root.title("Adaptive SHA-256 Field Solver")
        self.running = True
        self.paused = False
        self.training_mode = True  # Start in training mode
        self.setup_gui()
        self.update_loop()

    def setup_gui(self):
        control_frame = ttk.Frame(self.root)
        control_frame.pack(side=tk.TOP, fill=tk.X, padx=10, pady=5)
        ttk.Label(control_frame, text="Update Rate:").pack(side=tk.LEFT, padx=5)
        self.update_rate_slider = ttk.Scale(control_frame, from_=0.01, to=1.0, orient=tk.HORIZONTAL)
        self.update_rate_slider.set(0.1)
        self.update_rate_slider.pack(side=tk.LEFT, padx=5)
        ttk.Label(control_frame, text="Training Rate:").pack(side=tk.LEFT, padx=5)
        self.train_rate_slider = ttk.Scale(control_frame, from_=0.01, to=1.0, orient=tk.HORIZONTAL)
        self.train_rate_slider.set(0.1)
        self.train_rate_slider.pack(side=tk.LEFT, padx=5)
        self.pause_button = ttk.Button(control_frame, text="Pause", command=self.toggle_pause)
        self.pause_button.pack(side=tk.LEFT, padx=5)
        # Add a mode toggle button
        self.mode_button = ttk.Button(control_frame, text="Mode: Training", command=self.toggle_mode)
        self.mode_button.pack(side=tk.LEFT, padx=5)
        self.status_label = ttk.Label(control_frame, text="Status: Running (Training Mode)", font=("Helvetica", 12))
        self.status_label.pack(side=tk.LEFT, padx=20)
        
        # Create a figure with three subplots:
        self.fig = Figure(figsize=(15, 5), dpi=100)
        self.ax_current = self.fig.add_subplot(131, projection='3d')
        self.ax_current.set_title("Predicted SHA-256 Hash")
        self.ax_target = self.fig.add_subplot(132, projection='3d')
        self.ax_target.set_title("True SHA-256 Hash")
        self.ax_network = self.fig.add_subplot(133)
        self.ax_network.set_title("Adaptive Network Graph")
        self.canvas = FigureCanvasTkAgg(self.fig, master=self.root)
        self.canvas.get_tk_widget().pack(fill=tk.BOTH, expand=True)

    def update_loop(self):
        if not self.running:
            return
        if not self.paused:
            # Generate a new SHA problem (features and target hash field)
            features, target_field = self.task.generate_target()
            # Use the controller to get a prediction/update the smooth field
            current_field = self.controller.update(features)
            # In training mode, update weights; in test mode, only evaluate performance.
            if self.training_mode:
                loss = self.controller.train_step(features, target_field)
                mode_text = "Training"
            else:
                loss = self.controller.test_step(features, target_field)
                mode_text = "Testing"
            self.status_label.config(text=f"Status: Running ({mode_text} Mode) | Loss: {loss:.4f}")
            self.update_field_plots(current_field, target_field.numpy())
            self.update_network_plot()
            self.canvas.draw_idle()
        self.root.after(100, self.update_loop)

    def update_field_plots(self, current_field: np.ndarray, target_field: np.ndarray):
        # Update 3D surface plots for predicted and true SHA-256 hash fields.
        self.ax_current.clear()
        self.ax_target.clear()
        x = np.linspace(0, 1, self.controller.field_size)
        y = np.linspace(0, 1, self.controller.field_size)
        X, Y = np.meshgrid(x, y)
        self.ax_current.plot_surface(X, Y, current_field, cmap='plasma')
        self.ax_current.set_title("Predicted SHA-256 Hash")
        self.ax_current.set_zlim(0, 1)
        self.ax_target.plot_surface(X, Y, target_field, cmap='viridis')
        self.ax_target.set_title("True SHA-256 Hash")
        self.ax_target.set_zlim(0, 1)

    def update_network_plot(self):
        # Update the network graph showing unit positions and their connections.
        self.ax_network.clear()
        self.ax_network.set_title("Adaptive Network Graph")
        self.ax_network.set_xlim(0, self.controller.field_size)
        self.ax_network.set_ylim(0, self.controller.field_size)
        role_colors = {
            "undifferentiated": "gray",
            "sensory": "blue",
            "memory": "green",
            "integration": "orange",
            "output": "red"
        }
        # Draw connections (edges)
        plotted_edges = set()
        for unit in self.controller.system.units:
            x1, y1 = unit.position.physical_coords
            for neighbor in unit.neighbors:
                edge = tuple(sorted([id(unit), id(neighbor)]))
                if edge in plotted_edges:
                    continue
                plotted_edges.add(edge)
                x2, y2 = neighbor.position.physical_coords
                self.ax_network.plot([x1, x2], [y1, y2], color="black", linewidth=0.5, alpha=0.5)
        # Draw nodes with colors based on their roles.
        for unit in self.controller.system.units:
            x, y = unit.position.physical_coords
            color = role_colors.get(unit.role, "black")
            self.ax_network.scatter(x, y, color=color, s=50, edgecolors='k')
            self.ax_network.text(x, y, unit.role, fontsize=8, ha='center', va='center')

    def toggle_pause(self):
        self.paused = not self.paused
        self.pause_button.config(text="Resume" if self.paused else "Pause")

    def toggle_mode(self):
        self.training_mode = not self.training_mode
        mode_str = "Training" if self.training_mode else "Testing"
        self.mode_button.config(text=f"Mode: {mode_str}")
        self.status_label.config(text=f"Status: Running ({mode_str} Mode)")

    def on_closing(self):
        self.running = False
        self.root.destroy()

# -------------------------------
# Main: Create controller, task, and launch GUI.
# -------------------------------
def main():
    root = tk.Tk()
    # For the SHA-256 problem, we use a 16x16 field (256 outputs) and an input_dim of 10.
    field_size = 16
    controller = AdaptiveFieldController(field_size=field_size, n_units=10, input_dim=10)
    task = SHA256Task(field_size=field_size, input_dim=10)
    app = AdaptiveFieldVisualizerApp(root, controller, task)
    root.protocol("WM_DELETE_WINDOW", app.on_closing)
    root.mainloop()

if __name__ == '__main__':
    main()
